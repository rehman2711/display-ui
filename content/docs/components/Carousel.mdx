## API

### Carousel Props

<Steps>

<Step>
  <p className="font-semibold">For Carousel Component</p>
</Step>

<APITable
  data={[
    {
      attribute: "children",
      type: "ReactNode | ReactNode[]",
      description:
        "Slides to be rendered inside the carousel. Typically composed of CarouselItem components.",
      default: "-",
    },
    {
      attribute: "size",
      type: '"sm" | "md" | "lg"',
      description: "Controls the maximum width of the carousel container.",
      default: '"md"',
    },
    {
      attribute: "autoPlay",
      type: "boolean",
      description: "Enables automatic slide transitions at a fixed interval.",
      default: "false",
    },
    {
      attribute: "intervalMs",
      type: "number",
      description:
        "Time interval in milliseconds between automatic slide transitions when autoPlay is enabled.",
      default: "3000",
    },
    {
      attribute: "className",
      type: "string",
      description:
        "Additional CSS classes to customize the carousel container styling.",
      default: "-",
    },
  ]}
/>

  <p className="font-semibold">For Carousel Item</p>
</Step>

<APITable

      default: "-",
    },
    {
      attribute: "className",
      type: "string",
      description:
        "Additional CSS classes to customize the carousel item styling.",

    },
