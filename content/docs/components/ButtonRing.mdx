## API

### Button Props

<APITable
  data={[
    {
      attribute: "children",
      type: "ReactNode | ReactNode[]",
      description:
        "The contents of the component . Usually the name of button",
      default: "-",
    },
    {
      attribute: "className",
      type: "ReactNode | ReactNode[]",
      description: "The extra classes for user modification .",
      default: "-",
    },
    {
      attribute: "variant",
      type: "solid | outline | ghost | link",
      description: "The button appearance style.",
      default: "solid",
    },
    {
      attribute: "size",
      type: "sm | md | lg",
      description: "The button size on viewport.",
      default: "md",
    },
    ,
    {
      attribute: "RingOff",
      type: "custom (in number) (in ms)",
      description: "The time after the ring get turned off.",
      default: "-",
    },
    {
      attribute: "radius",
      type: "sm | md | lg | xl | full",
      description: "The button radius on viewport.",
      default: "md",
    }
  ]}
/>
