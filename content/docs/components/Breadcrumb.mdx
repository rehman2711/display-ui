## Usage

```tsx
import { Breadcrumb } from "@/components/display-ui/Breadcrumb"
```

```tsx
<Breadcrumb.Main separator={<ChevronRight width={14} />} textSize="xs">
  <Breadcrumb.Path icon={<House width={12} />}>Home</Breadcrumb.Path>
  <Breadcrumb.Path icon={<ShoppingCart width={12} />}>Cart</Breadcrumb.Path>
  <Breadcrumb.Path icon={<Receipt width={12} />}>Billing</Breadcrumb.Path>
  <Breadcrumb.Path icon={<Truck width={12} />}>Shipping</Breadcrumb.Path>
  <Breadcrumb.Path icon={<BadgeDollarSign width={12} />}>
    Payment
  </Breadcrumb.Path>
</Breadcrumb.Main>
```

## API

### Breadcrumb Props

<Steps>

<Step>
  <p className="font-semibold">For Breadcrumb Main</p>
</Step>

<APITable
  data={[
    {
      attribute: "children",
      type: "ReactNode | ReactNode[]",
      description:
        "Breadcrumb items to be rendered in sequence. Typically composed of Breadcrumb.Path components.",
      default: "-",
    },
    {
      attribute: "textSize",
      type: '"xs" | "sm" | "md" | "lg"',
      description: "Controls the text size of the breadcrumb items.",
      default: '"md"',
    },
    {
      attribute: "separator",
      type: "ReactNode",
      description: "Custom separator rendered between breadcrumb items.",
      default: "-",
    },
    {
      attribute: "className",
      type: "string",
      description:
        "Additional CSS classes to customize the breadcrumb container styling.",
      default: "-",
    },
  ]}
/>

<Step>
  <p className="font-semibold">For Breadcrumb Path</p>
</Step>

<APITable
  data={[
    {
      attribute: "children",
      type: "ReactNode",
      description: "The label text of the breadcrumb path.",
      default: "-",
    },
    {
      attribute: "icon",
      type: "ReactNode",
      description: "Optional icon displayed before the breadcrumb text.",
      default: "-",
    },
    {
      attribute: "href",
      type: "string",
      description: "URL the breadcrumb path links to.",
      default: "-",
    },
    {
      attribute: "className",
      type: "string",
      description:
        "Additional CSS classes to customize the breadcrumb path styling.",
      default: "-",
    },
  ]}
/>

</Steps>
