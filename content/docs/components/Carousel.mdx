## API

### Carousel Props

<APITable
  data={[
    {
      attribute: "children",
      type: "ReactNode | ReactNode[]",
      description: "The contents of the component . Usually the name of button",
      default: "-",
    },
    {
      attribute: "className",
      type: "ReactNode | ReactNode[]",
      description: "The extra classes for user modification .",
      default: "-",
    },
    {
      attribute: "autoPlay",
      type: "true | false",
      description: "Auto dragging of the slide / images .",
      default: "false",
    },
    {
      attribute: "intervalMs",
      type: "ms",
      description: "Time interval between two slide to be dragged.",
      default: "-",
    },
    {
      attribute: "size",
      type: "sm | md | lg",
      description: "The Size of carousel area",
      default: "-",
    },
  ]}
/>
